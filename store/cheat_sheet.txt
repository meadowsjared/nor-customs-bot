# 3. LOBBY & TEAM MANAGEMENT (lobby_players table)

# Get all currently active lobby players with roles, MMR, and teams:
SELECT
  lp.guild_id,
  lp.team,
  lp.lobby_rank,
  lp.role,
  p.discord_display_name,
  ha.hots_battle_tag,
  GREATEST(COALESCE(ha.hp_qm_mmr, 0), COALESCE(ha.hp_sl_mmr, 0), COALESCE(ha.hp_ar_mmr, 0)) + COALESCE(p.adjustment, 0) AS calculated_mmr
FROM lobby_players lp
JOIN players p ON lp.discord_id = p.discord_id
LEFT JOIN hots_accounts ha ON ha.discord_id = p.discord_id AND ha.is_primary = TRUE
WHERE lp.active = TRUE
ORDER BY lp.team NULLS LAST, lp.lobby_rank ASC;

# View configured bot channels:
SELECT guild_id, channel_type, channel_id, channel_name, updated_at FROM channels;

# View guild settings:

# View account loan substitutions:
SELECT id, guild_id, hots_battle_tag, original_discord_id, substitute_discord_id, date, is_active
FROM account_substitutions
ORDER BY date DESC;

# Check active map veto voting sessions and votes:
SELECT s.id AS session_id, s.title, s.created_by, s.active, v.user_name, v.map_id, v.voted_at
FROM map_vote_sessions s
LEFT JOIN map_votes v ON s.id = v.session_id
ORDER BY s.created_at DESC
LIMIT 50;

# 5. REPLAYS & PLAYER STATS (replays & mv_replay_player_stats)
# ==============================================================================

# Refresh the player stats materialized view (run after importing replays):
REFRESH MATERIALIZED VIEW CONCURRENTLY mv_replay_player_stats;

# Wipe replays table and reset view:
TRUNCATE TABLE replays CASCADE;
REFRESH MATERIALIZED VIEW mv_replay_player_stats;

# Get total replays count:
SELECT COUNT(*) FROM replays;

# Map popularity across all replays:
SELECT map, COUNT(*) AS games_played
FROM replays
GROUP BY map
ORDER BY games_played DESC;

# Recent match history by BattleTag:
SELECT
  date,
  map,
  hots_battle_tag,
  hero,
  CASE WHEN win THEN 'WIN' ELSE 'LOSS' END AS result,
  solo_kills AS kills,
  assists,
  deaths,
  ROUND(hero_damage::numeric) AS hero_damage,
  ROUND(siege_damage::numeric) AS siege_damage,
  ROUND(healing::numeric) AS healing
FROM mv_replay_player_stats
WHERE hots_battle_tag ILIKE '%Bandayd%'
ORDER BY date DESC
LIMIT 20;

# Recent match history by Discord Name / Display Name:
SELECT
  s.date,
  s.map,
  p.discord_display_name,
  s.hots_battle_tag,
  s.hero,
  CASE WHEN s.win THEN 'WIN' ELSE 'LOSS' END AS result,
  s.solo_kills AS kills,
  s.assists,
  s.deaths,
  ROUND(s.hero_damage::numeric) AS hero_damage,
  ROUND(s.siege_damage::numeric) AS siege_damage,
  ROUND(s.healing::numeric) AS healing
FROM mv_replay_player_stats s
JOIN hots_accounts ha ON LOWER(s.hots_battle_tag) = LOWER(ha.hots_battle_tag)
JOIN players p ON ha.discord_id = p.discord_id
WHERE p.discord_name ILIKE '%Bandayd%' OR p.discord_display_name ILIKE '%Bandayd%'
ORDER BY s.date DESC
LIMIT 20;

# Top sprayers with full BM / emote stats:
SELECT
  s.hots_battle_tag,
  p.discord_display_name,
  SUM(s.sprays_count) AS total_sprays,
  SUM(s.spray_td) AS spray_td,
  SUM(s.spray_deaths) AS spray_deaths,
  SUM(s.bsteps_count) AS total_bsteps,
  SUM(s.bstep_td) AS bstep_td,
  SUM(s.bstep_deaths) AS bstep_deaths,
  SUM(s.taunts_count) AS total_taunts,
  SUM(s.dances_count) AS total_dances
FROM mv_replay_player_stats s
LEFT JOIN hots_accounts ha ON LOWER(s.hots_battle_tag) = LOWER(ha.hots_battle_tag)
LEFT JOIN players p ON ha.discord_id = p.discord_id
GROUP BY s.hots_battle_tag, p.discord_display_name
ORDER BY total_sprays DESC
LIMIT 10;

# Top dancers with full BM / emote stats:
SELECT
  s.hots_battle_tag,
  p.discord_display_name,
  SUM(s.dances_count) AS total_dances,
  SUM(s.dance_td) AS dance_td,
  SUM(s.dance_deaths) AS dance_deaths,
  SUM(s.bsteps_count) AS total_bsteps,
  SUM(s.taunts_count) AS total_taunts,
  SUM(s.sprays_count) AS total_sprays
FROM mv_replay_player_stats s
LEFT JOIN hots_accounts ha ON LOWER(s.hots_battle_tag) = LOWER(ha.hots_battle_tag)
LEFT JOIN players p ON ha.discord_id = p.discord_id
GROUP BY s.hots_battle_tag, p.discord_display_name
ORDER BY total_dances DESC
LIMIT 10;

# Player overall win rates and performance (min 5 games):
SELECT
  s.hots_battle_tag,
  p.discord_display_name,
  COUNT(*) AS games,
  COUNT(*) FILTER (WHERE s.win) AS wins,
  ROUND((COUNT(*) FILTER (WHERE s.win)::numeric / COUNT(*)) * 100, 1) || '%' AS win_rate,
  ROUND(AVG(s.kda)::numeric, 2) AS avg_kda,
  ROUND(AVG(s.dpm)::numeric, 1) AS avg_dpm,
  SUM(s.bsteps_count) AS total_bsteps,
  SUM(s.sprays_count) AS total_sprays
FROM mv_replay_player_stats s
LEFT JOIN hots_accounts ha ON LOWER(s.hots_battle_tag) = LOWER(ha.hots_battle_tag)
LEFT JOIN players p ON ha.discord_id = p.discord_id
GROUP BY s.hots_battle_tag, p.discord_display_name
HAVING COUNT(*) >= 5
ORDER BY (COUNT(*) FILTER (WHERE s.win)::numeric / COUNT(*)) DESC
LIMIT 20;
